export function getBreadcrumbs() {
  return [
    { id: 'home', label: 'Реєстр Словників' },
    { id: 'current', label: dictionary.current()?.name || (getDictionary.loading ? 'Завантаження...' : '') },
  ];
}

export function onNavigate(id) {
  if (id === 'home') {
    dictionary.goToList();
  }
}
