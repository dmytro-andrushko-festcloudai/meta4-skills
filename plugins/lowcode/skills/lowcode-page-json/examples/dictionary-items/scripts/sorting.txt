export const selectedSort = state('');

export const options = [
  { value: 'name:asc', label: 'Сортувати за назвою від А-Я' },
  { value: 'name:desc', label: 'Сортувати за назвою від Я-А' },
];

export function getItems(selectedValue) {
  return options.map((option) => ({ ...option, selected: selectedValue === option.value }));
}

export function getOrderBy() {
  const [field, direction] = (MenuSorting.selectedValue || '').split(':');

  return field ? [{ [field]: `${direction}_nulls_last` }] : [{ audit_updated_at: 'desc_nulls_last' }];
}

export async function onSort() {
  if (MenuSorting.selectedValue === selectedSort) {
    MenuSorting.setSelectedValue('');
  }

  update(selectedSort, MenuSorting.selectedValue);

  await items.getData();
}
