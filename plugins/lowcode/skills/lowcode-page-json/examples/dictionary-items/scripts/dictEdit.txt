const CLEAR_DELAY = 100;

export const loading = state(false);

export function notBlank(field) {
  return !field?.required || Boolean(field.value?.trim());
}

function getFields() {
  return [InputEditDictName, InputEditDictCode, RadioEditDictExtensible, InputEditDictDescription];
}

export function toggleModal() {
  ModalEditDictionary.setOpen(true);
}

export async function close() {
  ModalEditDictionary.setOpen(false);

  await new Promise((resolve) => setTimeout(resolve, CLEAR_DELAY));

  if (ModalEditDictionary.open) return;

  getFields().forEach((field) => field?.reset());
}

export function getDisableSubmit() {
  const fields = getFields();

  return fields.some((field) => !field?.valid) || !fields.some((field) => Boolean(field?.dirty));
}

export async function onSubmit() {
  update(loading, true);

  try {
    const object = {
      name: InputEditDictName.value?.trim(),
      code: InputEditDictCode.value?.trim() || null,
      description: InputEditDictDescription.value?.trim() || null,
      is_extensible: RadioEditDictExtensible.value === 'yes',
    };

    const { data, error } = await updateDictionary.run({ id: constants.id, object });

    if (error) throw new Error(error);

    if (!data?.dictionary?.fcp_id) {
      throw new Error('Не вдалося оновити словник');
    }

    app.showAlert(`Словник "${data.dictionary.name}" успішно збережено`, 'success');

    close();

    await dictionary.reload();
  } catch (error) {
    app.showAlert(error?.message || 'Не вдалося оновити словник', 'error');
  } finally {
    update(loading, false);
  }
}
