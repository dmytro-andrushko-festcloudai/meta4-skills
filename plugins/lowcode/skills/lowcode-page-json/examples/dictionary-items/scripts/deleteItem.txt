const CLEAR_DELAY = 100;

export const item = state(null);
export const loading = state(false);

export function toggleModal(row) {
  update(item, row);
  ModalDeleteItem.setOpen(true);
}

export async function close() {
  ModalDeleteItem.setOpen(false);

  await new Promise((resolve) => setTimeout(resolve, CLEAR_DELAY));

  if (ModalDeleteItem.open) return;

  update(item, null);
}

export function getLabel() {
  return item?.name || '';
}

export function getDescription() {
  return `Значення "${getLabel()}" буде видалено назавжди. Його не можна буде відновити`;
}

export async function onSubmit() {
  update(loading, true);

  try {
    const { data, error } = await deleteDictionaryItem.run({ id: item?.fcp_id });

    if (error) throw new Error(error);

    if (!data?.item?.fcp_id) {
      throw new Error('Не вдалося видалити значення');
    }

    app.showAlert('Значення успішно видалено', 'success');

    close();

    await getDictionaryItems.run();
  } catch (error) {
    app.showAlert(error?.message || 'Не вдалося видалити значення', 'error');
  } finally {
    update(loading, false);
  }
}
