}

export function getWhere() {
  const search = InputSearch.value?.trim();
  const filters = [{ dictionary_fcp_id: { _eq: constants.id } }];

  if (search) {
    filters.push({
      _or: [{ name: { _ilike: `%${search}%` } }, { code: { _ilike: `%${search}%` } }],
    });
  }

  filters.push(...filter.getConditions());

  return { _and: filters };
}
