const ALPHABET = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ'.split('');

export const selectedFilters = state({ name: [], status: [] });

function widgetsOf(id) {
  const groups = { name: CheckboxGroupName, status: CheckboxGroupStatus };
  const searches = { name: InputFilterName };

  return { group: groups[id], search: searches[id] };
}

export function getLetterOptions(id) {
  const search = widgetsOf(id).search?.value?.trim().toUpperCase() || '';

  return ALPHABET.filter((letter) => letter.includes(search)).map((letter) => ({ value: letter, label: letter }));
}

export function getStatusOptions() {
  return [
    { value: 'active', label: 'Активний' },
    { value: 'inactive', label: 'Не активний' },
  ];
}

export function onLetterSearch() {}

function getDraft() {
  return {
    name: CheckboxGroupName.values || [],
    status: CheckboxGroupStatus.values || [],
  };
}

export function getConditions() {
  const { name, status } = selectedFilters;
  const conditions = [];

  if (name.length > 0) {
    conditions.push({ name: { _iregex: `^[${name.join('')}]` } });
  }

  if (status.length === 1) {
    conditions.push({ is_active: { _eq: status[0] === 'active' } });
  }

  return conditions;
}

export function onClose() {
  CheckboxGroupName.setValues(selectedFilters.name);
  CheckboxGroupStatus.setValues(selectedFilters.status);
}

export function onApply() {
  update(selectedFilters, getDraft());

  items.getData();
}

export function onClearCategory(id) {
  const { group, search } = widgetsOf(id);

  group?.reset();

  if (search?.value) search.setValue('');
}

export function onClear() {
  ['name', 'status'].forEach(onClearCategory);
}

export function onQuickClear() {
  onClear();

  update(selectedFilters, { name: [], status: [] });

  items.getData();
}

export function isActive() {
  return Object.values(selectedFilters).some((values) => values.length > 0);
}

export function selectedQty() {
  const total = Object.values(selectedFilters).reduce((sum, values) => sum + values.length, 0);

  return total > 99 ? '99+' : total;
}

export function isEmpty() {
  return Object.values(getDraft()).every((values) => values.length === 0);
}

export function isChanged() {
  const draft = getDraft();
  const isSame = (a, b) => a.length === b.length && a.every((value) => b.includes(value));

  return Object.keys(selectedFilters).some((key) => !isSame(draft[key], selectedFilters[key]));
}
