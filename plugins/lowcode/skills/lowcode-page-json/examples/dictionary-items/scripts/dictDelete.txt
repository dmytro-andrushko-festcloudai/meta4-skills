export const loading = state(false);

export function toggleModal() {
  ModalDeleteDictionary.setOpen(true);
}

export function close() {
  ModalDeleteDictionary.setOpen(false);
}

function getItemsCount() {
  return dictionary.current()?.total?.aggregate?.count ?? 0;
}

function getItemsLabel(count) {
  const lastDigit = count % 10;
  const lastTwoDigits = count % 100;

  if (lastDigit >= 1 && lastDigit <= 4 && (lastTwoDigits < 11 || lastTwoDigits > 14)) {
    return `${count} значення`;
  }

  return `${count} значень`;
}

export function getDescription() {
  const name = dictionary.current()?.name || '';
  const count = getItemsCount();

  if (!count) {
    return `Словник "${name}" буде видалено назавжди. Його не можна буде відновити`;
  }

  return `Словник "${name}" та ${getItemsLabel(count)} у ньому будуть видалені назавжди. Їх не можна буде відновити`;
}

export async function onSubmit() {
  update(loading, true);

  try {
    const { error: itemsError } = await deleteDictionaryItems.run({ id: constants.id });

    if (itemsError) throw new Error(itemsError);

    const { data, error } = await deleteDictionary.run({ id: constants.id });

    if (error) throw new Error(error);

    if (!data?.dictionary?.fcp_id) {
      throw new Error('Не вдалося видалити словник');
    }

    app.showAlert('Словник успішно видалено', 'success');

    close();
    dictionary.goToList();
  } catch (error) {
    app.showAlert(error?.message || 'Не вдалося видалити словник', 'error');
  } finally {
    update(loading, false);
  }
}
