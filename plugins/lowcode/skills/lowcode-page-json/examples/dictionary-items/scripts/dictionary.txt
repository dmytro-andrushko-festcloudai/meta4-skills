export function current() {
  return getDictionary.data?.dictionaries?.[0] || null;
}

export function canAddItems() {
  return Boolean(current()?.is_extensible);
}

export function goToList() {
  app.navigate(constants.basePath);
}

export async function load() {
  if (!constants.id) return;

  const { error } = await getDictionary.run({ id: constants.id });

  if (error) {
    app.showAlert(error, 'error');

    return;
  }

  await items.getData();
}

export async function reload() {
  const { error } = await getDictionary.run({ id: constants.id });

  if (error) app.showAlert(error, 'error');
}
