return {
  getDictionary() {
    return getDictionary.data?.dictionaries?.[0] || null;
  },

  canAddItems() {
    return Boolean(this.getDictionary()?.is_extensible);
  },

  goToList() {
    app.navigate(constants.basePath);
  },

  async reload() {
    const { error } = await getDictionary.run({ id: init.id });

    if (error) app.showAlert(error, 'error');
  },
};
