export const selectedFilters = state({ names: [], manufacturers: [], countries: [] });

function widgetsOf(id) {
  const groups = { name: CheckboxGroupName, manufacturer: CheckboxGroupManufacturer, country: CheckboxGroupCountry };
  const searches = { name: InputFilterName, manufacturer: InputFilterManufacturer, country: InputFilterCountry };

  return { group: groups[id], search: searches[id] };
}

export function getSearchVariables() {
  const toPattern = (input) => `%${input.value?.trim() || ''}%`;

  return {
    nameSearch: toPattern(InputFilterName),
    manufacturerSearch: toPattern(InputFilterManufacturer),
    countrySearch: toPattern(InputFilterCountry),
  };
}

function toOptions(rows, key) {
  return (rows || [])
    .filter((row) => row[key]?.trim())
    .map((row) => ({ value: row[key], label: row[key].trim() }));
}

export function getNameOptions() {
  return toOptions(getFilters.data?.names, 'name');
}

export function getManufacturerOptions() {
  return toOptions(getFilters.data?.manufacturers, 'manufacturer_name');
}

export function getCountryOptions() {
  return toOptions(getFilters.data?.countries, 'country_of_origin');
}

export function onSearch() {
  getFilters.run();
}

function getDraft() {
  return {
    names: CheckboxGroupName.values || [],
    manufacturers: CheckboxGroupManufacturer.values || [],
    countries: CheckboxGroupCountry.values || [],
  };
}

export function onClose() {
  CheckboxGroupName.setValues(selectedFilters.names);
  CheckboxGroupManufacturer.setValues(selectedFilters.manufacturers);
  CheckboxGroupCountry.setValues(selectedFilters.countries);
}

export function onApply() {
  update(selectedFilters, getDraft());

  table.getData();
}

export function onClearCategory(id) {
  const { group, search } = widgetsOf(id);

  group?.reset();

  if (search?.value) {
    search.setValue('');
    getFilters.run();
  }
}

export function onClear() {
  ['name', 'manufacturer', 'country'].forEach(onClearCategory);
}

export function onQuickClear() {
  onClear();

  update(selectedFilters, { names: [], manufacturers: [], countries: [] });

  table.getData();
}

export function isActive() {
  return Object.values(selectedFilters).some((values) => values.length > 0);
}

export function selectedQty() {
  const total = Object.values(selectedFilters).reduce((sum, values) => sum + values.length, 0);

  return total > 99 ? '99+' : total;
}

export function isEmpty() {
  return Object.values(getDraft()).every((values) => values.length === 0);
}

export function isChanged() {
  const draft = getDraft();
  const isSame = (a, b) => a.length === b.length && a.every((value) => b.includes(value));

  return Object.keys(selectedFilters).some((key) => !isSame(draft[key], selectedFilters[key]));
}
