export function getBreadcrumbs() {
  return [
    { id: 'home', label: 'Менеджер таксономії' },
    { id: 'merge', label: 'Об’єднання таксономії' },
  ];
}

export function onNavigate(id) {
  if (id === 'home') {
    app.navigate(constants.basePath);
  }
}

export function onBack() {
  app.navigate(constants.basePath);
}
