export const loading = state(false);
export const taxonomies = state([]);

export const itemsById = Object.fromEntries(taxonomies.map((item) => [item.fcp_id, item]));

export const childrenOf = taxonomies
  .filter((item) => item.parent_fcp_id && itemsById[item.parent_fcp_id])
  .reduce((result, item) => ({ ...result, [item.parent_fcp_id]: [...(result[item.parent_fcp_id] || []), item] }), {});

export const rootIds = taxonomies
  .filter((item) => !item.parent_fcp_id || !itemsById[item.parent_fcp_id])
  .map((item) => item.fcp_id);

export async function reload({ notify = true } = {}) {
  update(loading, true);

  try {
    const { data, error } = await getTaxonomies.run();

    if (error) throw new Error(error);

    update(taxonomies, data?.taxonomies || []);

    return true;
  } catch (error) {
    if (notify) {
      app.showAlert(error?.message || 'Не вдалося завантажити таксономії', 'error');
    }

    return false;
  } finally {
    update(loading, false);
  }
}

export function getNodes(parentId) {
  if (!parentId) return [];

  return (childrenOf[parentId] || []).map((item) => {
    const children = getNodes(item.fcp_id);
    const node = { id: item.fcp_id, label: item.name, item, icon: children.length > 0 ? 'BookOpen' : 'Tag' };

    if (children.length > 0) {
      node.children = children;
    }

    return node;
  });
}

export function flatten(nodes) {
  return nodes.flatMap((node) => [node, ...flatten(node.children || [])]);
}

export function findNode(nodes, id) {
  if (!id) return null;

  return flatten(nodes).find((node) => node.id === id) || null;
}

function countNodes(parentId) {
  return (childrenOf[parentId] || []).reduce((sum, item) => sum + 1 + countNodes(item.fcp_id), 0);
}

export function getPath(id) {
  const path = [];
  let current = itemsById[id];

  while (current) {
    path.unshift(current.fcp_id);
    current = itemsById[current.parent_fcp_id];
  }

  return path;
}

export function getRootOptions(excludeId) {
  return rootIds
    .filter((id) => id !== excludeId)
    .map((id) => {
      const { name, type } = itemsById[id];
      const label = type ? `${name} (${type})` : name;

      return { value: id, label: `${label} - ${countNodes(id)} nodes` };
    })
    .sort((a, b) => a.label.localeCompare(b.label));
}
