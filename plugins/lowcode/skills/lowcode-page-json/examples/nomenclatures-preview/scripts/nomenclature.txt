export function getItem() {
  return getNomenclature.data?.nomenclatures?.[0] || null;
}

export async function load() {
  if (!constants.id) return;

  const { error } = await getNomenclature.run({ id: constants.id });

  if (error) {
    app.showAlert(error, 'error');
  }
}

export function text(value) {
  if (value === null || value === undefined) return '—';

  return String(value).trim() || '—';
}

export function yesNo(value) {
  if (value === null || value === undefined) return '—';

  return value ? 'Так' : 'Ні';
}

export function temperature(value) {
  const formatted = text(value);

  return /^[-+]?\d+([.,]\d+)?$/.test(formatted) ? `${formatted} (°C)` : formatted;
}
