export function getBreadcrumbs() {
  const home = { id: 'home', label: 'Товари' };

  if (!constants.id) {
    return [home, { id: 'current', label: 'Створення товару' }];
  }

  return [
    home,
    { id: constants.id, label: form.getItem()?.name || 'Завантаження...' },
    { id: 'current', label: 'Редагування товару' },
  ];
}

export function onNavigate(id) {
  if (id === 'home') {
    app.navigate(constants.basePath);
  } else if (id !== 'current') {
    app.navigate(`${constants.basePath}/${id}`);
  }
}

export function onBack() {
  app.navigate(constants.id ? `${constants.basePath}/${constants.id}` : constants.basePath);
}
